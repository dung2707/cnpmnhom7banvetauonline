}: Props) {

    <div
      className="train-carriage-viewport"

    >
      {cars.map((car) => (
        <button
          type="button"
          key={car.number}
          aria-pressed={selectedCarNumber === car.number}
          aria-label={`${car.title}, ${car.typeLabel}, còn ${car.availableSeatCount} chỗ`}
          onClick={() => onSelectCar(car.number)}
        >
          Toa {car.number}
        </button>
      ))}
    </div>
  )
}

function renderSeat(seat: SeatRow) {
  const isSelected =
    selectedPlaces.some((place) => place.id === seat.id)

  const ownHold =
    seatHolds.some(
      (hold) =>
        hold.seat_id === seat.id &&
        hold.session_id === sessionId,
    )

  const isHeldElsewhere =
    seat.status === 'held' && !ownHold

  const isUnavailable =
    seat.status === 'booked' || isHeldElsewhere

  return (
    <button
      className={[
        'seat-tile',
        isSelected ? 'seat-tile-selected' : '',
        seat.status === 'booked' ? 'seat-tile-sold' : '',
        isHeldElsewhere ? 'seat-tile-held' : '',
      ].join(' ')}
      type="button"
      key={seat.id}
      disabled={isUnavailable || seatLoading}
